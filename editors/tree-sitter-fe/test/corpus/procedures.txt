procedure CONTROLS {
    name     "Controls"
    category normal

    check FUEL_QUANTITY_INDICATION
    set FUEL_XFEED_SELECTOR = TANK_3_TO_1
    set FUEL_PUMP_PRESSURE_TARGET = 22.5
    start HYD_2_ELECTRIC_PUMP
    stop HYD_2_ELECTRIC_PUMP
    open FUEL_CROSSFEED_VALVE
    close FUEL_CROSSFEED_VALVE
    notify "Crossfeed complete"
    call ELEC_BUS_2_RESTORE
}

procedure WAITING {
    name     "Waiting"
    category normal

    wait hydraulic.2.pressure > 2500
    wait hydraulic.2.pressure > 2500 timeout 30s
    wait hydraulic.2.pressure > 2500 timeout 500ms else fail
    complete
    complete when cabin.altitude < 10000 timeout 5m
    fail
    fail "system 2 pressure not restored"
}

procedure BRANCHING {
    name     "Branching"
    category normal

    if generator.2.available {
        start GEN_2_FIELD
    } else if electrical.ac_bus.1.powered {
        set BUS_TIE_1_2 = CLOSED
    } else {
        fail "no power source"
    }
}
